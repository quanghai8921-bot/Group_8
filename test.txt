2️⃣ Sinh 10.000 dữ liệu mẫu
WITH RECURSIVE seq AS (
    SELECT 1 AS n
    UNION ALL
    SELECT n + 1 FROM seq WHERE n < 10000
)
INSERT INTO shopeefood_orders (
    order_id,
    user_id,
    restaurant_id,
    restaurant_name,
    food_name,
    quantity,
    order_value,
    delivery_fee,
    order_time,
    status,
    rating
)
SELECT
    n AS order_id,
    FLOOR(1 + RAND() * 2000) AS user_id,
    FLOOR(1 + RAND() * 300) AS restaurant_id,
    CONCAT('Quán ăn ', FLOOR(1 + RAND() * 300)) AS restaurant_name,
    ELT(FLOOR(1 + RAND() * 6),
        'Trà sữa',
        'Cơm tấm',
        'Bún bò',
        'Pizza',
        'Gà rán',
        'Mì xào'
    ) AS food_name,
    FLOOR(1 + RAND() * 5) AS quantity,
    ROUND(30000 + RAND() * 150000, 0) AS order_value,
    ROUND(10000 + RAND() * 30000, 0) AS delivery_fee,
    NOW() - INTERVAL FLOOR(RAND() * 365) DAY AS order_time,
    ELT(FLOOR(1 + RAND() * 4),
        'Completed',
        'Cancelled',
        'Delivering',
        'Pending'
    ) AS status,
    FLOOR(1 + RAND() * 5) AS rating
FROM seq;

Bạn đang chạy file mình tạo bằng SQL Server, nhưng file đó là MySQL nên SQL Server báo lỗi. Cụ thể:

SET NAMES ...: chỉ có ở MySQL ⇒ SQL Server báo 'NAMES' is not a recognized SET option.

SET FOREIGN_KEY_CHECKS=0/1: MySQL thôi ⇒ SQL Server báo lỗi cú pháp.

DROP TABLE IF EXISTS ...: SQL Server không hỗ trợ cú pháp này theo kiểu MySQL (tuỳ phiên bản) ⇒ lỗi “Incorrect syntax near '(' …” dây chuyền.

SQL Server giới hạn INSERT ... VALUES (...) , (...) tối đa 1000 dòng/statement ⇒ lỗi exceeds the maximum allowed number of 1000 row values (file mình có chỗ batch > 1000).

Lỗi gần dấu = cuối file là do SET FOREIGN_KEY_CHECKS=1 (SQL Server không hiểu).

Mình đã chuyển toàn bộ sang T-SQL chuẩn cho SQL Server + chia batch INSERT tối đa 1000 dòng để import chạy được.
