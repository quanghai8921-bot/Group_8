	Full_Name nvarchar(50) NOT NULL,
	Birth_Date DATE NOT NULL,

CREATE TABLE User_Role_Mapping(

CREATE TABLE Merchant(
	Merchant_ID varchar(10) PRIMARY KEY,
	[User_ID] varchar(10) NOT NULL UNIQUE,
	Store_Name nvarchar(100) NOT NULL,
	Store_Address nvarchar(100) NOT NULL,

	CONSTRAINT FK_Merchant_User FOREIGN KEY ([User_ID]) REFERENCES [User]([User_ID])

CREATE TABLE Driver(
	Driver_ID varchar(10) PRIMARY KEY,
	[User_ID] varchar(10) NOT NULL UNIQUE,

	CONSTRAINT FK_Location_Driver FOREIGN KEY (Driver_ID) REFERENCES Driver(Driver_ID)

CREATE TABLE Menu_Category(

Để đảm bảo dữ liệu "sạch" và logic như bạn yêu cầu, tôi sẽ thực hiện theo quy trình sau:

Giai đoạn 1: Dữ liệu nền tảng (User & Role)

Giai đoạn 2: Merchant & Food (Hệ sinh thái quán ăn)

Giai đoạn 3: Driver

Lấy từ tập User có role Driver.

Biển số xe: Random theo format 59-X1 123.45.

Giai đoạn 4: Order Flow (Luồng đơn hàng - Quan trọng nhất)

Quy trình ngược: Tôi sẽ tạo Order trước, sau đó random món ăn (Order_Detail) thuộc quán đó.

Tính toán: Tổng tiền món = Sum(Đơn giá * Số lượng).

Thành tiền = Tổng tiền món + Ship - Voucher.

Bảng Payment sẽ lấy chính xác con số Thành tiền này.
